# Identity operators

# is - check for same id (memory reference)
# is not - check for different id (memory reference)
# evry value in python is an object and every object has an identity (id) which is unique and constant for this object during its lifetime. The id of an object can be obtained using the built-in id() function.
# objcts properties : 1. value 2. type 3. identity(memory address)

# *Immutable* objects (like int, float, string, tuple) with the same value may
# have the *same id (memory reference) due to interning,
# while *mutable* objects (like list, dict, set) will always
# have *different ids even if they have the same content.

# immutable
a = 10
b = 10
print(id(a))
print(id(b))
print(a is b)   # True

# mutable
a = [1, 2, 3]
b = [1, 2, 3]
print(id(a))
print(id(b))
print(a is b)  # False


# logical Operators

def student_eligibility_checker(age, attendance, marks, fee):
    # AND
    if age > 18 and attendance >= 75 and fee ==  True:
        print("You are eligible for the exam")
    else:
        print("You are not eligible for the exam")
 
    #OR
    if attendance >= 90 or marks >= 95:
        print("You are eligible for the scholarship")
    else:
        print("You are not eligible for the scholarship")
 
    # NOT
    if not fee:
        print("Please pay the exam fee)")
 
    # XOR
    if marks >= 95 ^ attendance >= 95:
        print("You are eligible for the special award")
    else:
        print("You are not eligible for the special award")

    marks = int(input("Marks: "))
    fee = input("Fee Paid (True/False): ") == 'True'
    student_eligibility_checker(age, attendance, marks, fee)

# Control Statements

# Sometimes we need to control the flow of our program. We can use control statements to do this.
# control statements allow us to:
# execute something only if a condition is true.
# repeat something many times.
# stop execution early.
# skip some statements.
# need to break the loop for a condition

# decision making statements
# if
a=10
if a == 19:
    print("prime number")

# if-else
rain = True
if rain:
    print("Take an umbrella")
else:
    print("No need for an umbrella")

# if-elif-else
age = 18
if age < 18:
    print("Young")
elif age >= 18 and age < 60:
    print("adult")
else:
    print("senior citizen")

# nested if
grade = 'A'
marks = 95
if grade == 'A':
    if marks >= 90:
        print("Excellent")
    else:
        print("Good")


# loop statements (for, while)

n = int(input("Enter a number: "))
sum = 0
for i in range(1, n+1, 2): # repeat fixed number of times
    sum += i
print(f"Sum = {sum}")

P = 10
while P > 0: # repeat until condition is true
    print(P)
    P -= 1


# Jump statements (break, continue, pass)
for i in range(1, 100, 25):
    if i == 26:
        continue
    if i == 51:
        break
    print(i)
